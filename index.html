Java Servlet Code 
import java.io.*;
import javax.servlet.*;
import javax.servlet.http.*;

public class ElectricityBillServlet extends HttpServlet {

    protected void doPost(HttpServletRequest request,
                          HttpServletResponse response)
            throws ServletException, IOException {

        // Read parameters from form
        String name = request.getParameter("consumerName");
        String consumerNo = request.getParameter("consumerNo");
        String meterNo = request.getParameter("meterNo");
        int prevReading = Integer.parseInt(
            request.getParameter("prevReading"));
        int currReading = Integer.parseInt(
            request.getParameter("currReading"));
        String category = request.getParameter("category");

        int units = currReading - prevReading;

        // Calculate energy charge based on slabs
        double energyCharge = 0;
        if (category.equals("residential")) {
            energyCharge = calcResidential(units);
        } else if (category.equals("commercial")) {
            energyCharge = calcCommercial(units);
        } else {
            energyCharge = calcIndustrial(units);
        }

        double fixedCharge = getFixedCharge(category);
        double duty = energyCharge * 0.10;
        double fuelSurcharge = energyCharge * 0.02;
        double total = energyCharge + fixedCharge
                       + duty + fuelSurcharge;

        // Send response
        response.setContentType("text/html");
        PrintWriter out = response.getWriter();
        out.println("<h2>Electricity Bill</h2>");
        out.println("<p>Name: " + name + "</p>");
        out.println("<p>Units: " + units + "</p>");
        out.println("<p>Total: ₹" +
            String.format("%.2f", total) + "</p>");
        out.close();
    }

    private double calcResidential(int u) {
        double c = 0;
        if (u <= 100) c = u * 3.50;
        else if (u <= 300) c = 100*3.50 + (u-100)*5.50;
        else if (u <= 500) c = 100*3.50 + 200*5.50
                              + (u-300)*7.00;
        else c = 100*3.50 + 200*5.50 + 200*7.00
                 + (u-500)*8.50;
        return c;
    }

    private double calcCommercial(int u) {
        return u * 8.00;
    }

    private double calcIndustrial(int u) {
        return u * 7.50;
    }

    private double getFixedCharge(String cat) {
        switch (cat) {
            case "residential": return 50;
            case "commercial": return 200;
            case "industrial": return 350;
            default: return 50;
        }
    }
}
